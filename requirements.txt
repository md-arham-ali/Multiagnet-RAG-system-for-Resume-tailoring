# =============================================================================
# RUNTIME dependencies: what the backend imports while it serves. This is the list the
# Docker image installs (together with requirements.lock for exact versions).
# Target interpreter: CPython 3.12 (3.14 lacks wheels for chromadb).
#
# Three files, three jobs:
#   requirements.txt     what the app needs, by name (this file; edit it when you add an import)
#   requirements.lock    the exact versions of this list and everything it pulls in (the image uses it)
#   requirements-ci.txt  what the test suite needs on GitHub Actions (pinned, no PyTorch)
#
# PyTorch is NOT pinned here. sentence-transformers pulls it in; the Dockerfile installs the
# CPU-only build first, from https://download.pytorch.org/whl/cpu, so no multi-GB CUDA wheel.
# Checked against the code on 2026-10-25 by scanning every import under agents/ backend/
# utils/ vectordb/ evaluation/ learning/ state/ config.py scripts/frontend_api.py.
# =============================================================================

# ---- Orchestration -----------------------------------------------------------
langgraph                       # the graph, supervisor and gates
langgraph-checkpoint-sqlite     # durable checkpointer (logs/checkpoints.sqlite)

# ---- LLM providers -----------------------------------------------------------
langchain-core
langchain-groq                  # Groq Llama
langchain-google-genai          # Gemini
langchain-openai                # OpenAI-compatible endpoints
google-genai                    # Gemini errors, imported by agents/base.py
langchain-text-splitters        # vectordb/pdf_ingest.py
google-api-core                 # Gemini errors, imported by agents/base.py

# ---- Validation, config, prompts ---------------------------------------------
pydantic
python-dotenv                   # config.py is the only reader of .env
PyYAML                          # versioned prompts

# ---- Knowledge base: vector store, embeddings, matching ----------------------
chromadb
sentence-transformers           # bge embedder + reranker (pulls in PyTorch)
numpy
rapidfuzz                       # keyword channel, ATS coverage, do-not-claim matching
pypdf                           # CV upload and parsing

# ---- Serving -----------------------------------------------------------------
fastapi
uvicorn                         # plain, without the [standard] extras
python-multipart                # FastAPI multipart/form-data (CV upload, backend/routes/cv.py)

# ---- Observability -----------------------------------------------------------
langfuse                        # utils/observability.py
